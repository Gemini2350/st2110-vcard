cmake_minimum_required(VERSION 3.16)
project(st2110-vcard VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()

option(ENABLE_NMOS "Build the embedded NMOS IS-04/IS-05 node (needs nmos-cpp)" ON)
option(VCARD_MOCK_MTL "Link a software loopback instead of libmtl (tests, no NIC needed)" OFF)
set(MTL_SOURCE_DIR "" CACHE PATH "Media-Transport-Library checkout; only needed with VCARD_MOCK_MTL when MTL is not installed")

find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)

if(VCARD_MOCK_MTL)
  # Headers only: the real ones, so the mock cannot drift from the API. Taken
  # from an installed MTL if there is one, else straight from a source checkout.
  add_library(vcard-mtl INTERFACE)
  pkg_check_modules(MTL QUIET IMPORTED_TARGET mtl)
  if(MTL_FOUND)
    target_include_directories(vcard-mtl INTERFACE ${MTL_INCLUDE_DIRS})
  elseif(MTL_SOURCE_DIR)
    set(_inc ${CMAKE_BINARY_DIR}/mtl-headers)
    file(MAKE_DIRECTORY ${_inc})
    file(CREATE_LINK ${MTL_SOURCE_DIR}/include ${_inc}/mtl SYMBOLIC)
    # generated by MTL's own build; only the version macros are needed here
    file(WRITE ${_inc}/mtl_build_config.h
      "#define MTL_VERSION_MAJOR 0\n#define MTL_VERSION_MINOR 0\n#define MTL_VERSION_LAST 0\n#define MTL_VERSION_EXTRA \"mock\"\n")
    target_include_directories(vcard-mtl INTERFACE ${_inc})
  else()
    message(FATAL_ERROR "VCARD_MOCK_MTL needs MTL headers: install MTL or pass -DMTL_SOURCE_DIR=<checkout>")
  endif()
  message(STATUS "MTL: MOCK loopback (no network I/O)")
else()
  # Media Transport Library exports a pkg-config file "mtl"
  pkg_check_modules(MTL REQUIRED IMPORTED_TARGET mtl)
  add_library(vcard-mtl INTERFACE)
  target_link_libraries(vcard-mtl INTERFACE PkgConfig::MTL)
endif()

# swr_alloc_set_opts2() and the AVChannelLayout API landed in FFmpeg 5.1
pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET
  libavformat>=59.27 libavcodec>=59.37 libavutil>=57.28
  libswscale>=6.7 libswresample>=4.7)

# Both are header-only. Use what the system has (works offline); fetch only
# what is missing. httplib needs path parameters, i.e. 0.14 or newer.
include(FetchContent)
find_package(nlohmann_json 3.10 QUIET)
if(NOT nlohmann_json_FOUND)
  FetchContent_Declare(json
    GIT_REPOSITORY https://github.com/nlohmann/json.git
    GIT_TAG v3.11.3 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(json)
endif()
find_package(httplib 0.14 QUIET)
if(NOT httplib_FOUND)
  FetchContent_Declare(httplib
    GIT_REPOSITORY https://github.com/yhirose/cpp-httplib.git
    GIT_TAG v0.18.3 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(httplib)
endif()

add_executable(st2110-vcard
  src/main.cpp
  src/config.cpp
  src/core/log.cpp
  src/core/formats.cpp
  src/core/mtl-device.cpp
  src/media/decoder.cpp
  src/media/encoder.cpp
  src/player/player.cpp
  src/recorder/recorder.cpp
  src/monitor/preview.cpp
  src/monitor/meters.cpp
  src/api/http-server.cpp
  src/engine.cpp
)

target_include_directories(st2110-vcard PRIVATE src)
if(VCARD_MOCK_MTL)
  target_sources(st2110-vcard PRIVATE tests/mock-mtl/mock_mtl.cpp)
endif()
target_link_libraries(st2110-vcard PRIVATE
  vcard-mtl
  PkgConfig::FFMPEG
  nlohmann_json::nlohmann_json
  httplib::httplib
  Threads::Threads
)

if(ENABLE_NMOS)
  # nmos-cpp's installed config links nmos-cpp::websocketpp against
  # websocketpp::websocketpp but does not look for it itself when nmos-cpp was
  # built against system packages; without this the generate step fails.
  find_package(websocketpp QUIET)
  find_package(nmos-cpp CONFIG QUIET)
  if(nmos-cpp_FOUND)
    target_sources(st2110-vcard PRIVATE src/nmos/nmos-node.cpp)
    target_link_libraries(st2110-vcard PRIVATE nmos-cpp::nmos-cpp)
    target_compile_definitions(st2110-vcard PRIVATE HAVE_NMOS)
    message(STATUS "NMOS node: enabled")
  else()
    message(WARNING "nmos-cpp not found - building without the NMOS node")
  endif()
endif()

install(TARGETS st2110-vcard RUNTIME DESTINATION bin)
install(DIRECTORY web/ DESTINATION share/st2110-vcard/web)
install(FILES config/vcard.example.json DESTINATION share/st2110-vcard)
